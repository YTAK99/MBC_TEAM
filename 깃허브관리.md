# 1. main 최신화
- git checkout develop
- git pull origin develop

# 2. 내 브랜치 이동
- git checkout 내브랜치이름

# 3. main 내용 내 브랜치에 반영
- git merge develop

# 4. 작업
- 코드 수정

# 5. 작업 후 커밋
- git add .
- git commit -m "변경 사항"

# 6. 내 브랜치 푸시(원격 저장소에 업로드)
- git push origin 내브랜치이름
